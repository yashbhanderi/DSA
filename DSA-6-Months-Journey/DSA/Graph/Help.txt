-- Build Adj List

public static List<int>[] BuildAdjList(int[][] graph)
{
    int n = graph.Length;

    List<int>[] adj = new List<int>[n];

    for (int i = 0; i < n; i++)
    {
        adj[i] = new List<int>();

        for (int j = 0; j < n; j++)
        {
            if (graph[i][j] == 1)
                adj[i].Add(j);
        }
    }

    return adj;
}
